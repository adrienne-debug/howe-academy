rules_version = '2';
// Firebase Storage rules for the Envirothon class site (project howe-envirothon) — CPPHOTOUP, 2026-11-10.
// Paste ALL of this into Console → Build → Storage → Rules → Publish (envirothon/SITE_SETUP.md §7).
// The only files here are the class photos the teacher or Adrienne add with ＋ Add photo ▸ 📷 My photo:
//   classPhotos/<classId>/<itemKey>/<photoId>.jpg — shrunk on the device first (longest side 1280 px, JPEG), usually 100–400 KB.
// Storage rules cannot read the Realtime Database, so they cannot check that the signer is that class's teacher or an admin:
// any signed-in login of this project could technically upload a JPEG under a class path. What the app SHOWS is the database
// record (classPhotos/<cid>/<item>/<photoId> in SITE_RULES.json), which only the class's teacher or an admin can write.
// Delete has no size or type check (there is no new file to check): Undo removes the file it just uploaded.
// Everything else in the bucket: denied.
service firebase.storage {
  match /b/{bucket}/o {
    match /classPhotos/{cid}/{item}/{file} {
      allow read: if request.auth != null;
      allow create, update: if request.auth != null
                            && request.resource.size < 3 * 1024 * 1024
                            && request.resource.contentType.matches('image/jpeg');
      allow delete: if request.auth != null;
    }
  }
}
